### Step 1: `route.ts` — Fix `fetchKPIs` query
```sql
-- Deduplicate to one risk per L3+Region (highest VAR), then aggregate
WITH deduped AS (
  SELECT CATEGORY_L3, REGION, DEPARTMENT, VALUE_AT_RISK_USD, UNITS_AT_RISK, 
    STOCKOUT_RATE_PCT, RISK_SEVERITY
  FROM FACT_DEMAND_RISK
  WHERE IS_CURRENT_WEEK = TRUE
  QUALIFY ROW_NUMBER() OVER (PARTITION BY CATEGORY_L3, REGION ORDER BY VALUE_AT_RISK_USD DESC) = 1
)
SELECT 
  COUNT(*) as TOTAL_ANOMALIES,  -- distinct L3+Region combos
  COUNT(CASE WHEN RISK_SEVERITY IN ('CRITICAL','HIGH') THEN 1 END) as HIGH_IMPACT,
  COUNT(DISTINCT DEPARTMENT) as DEPARTMENTS_AFFECTED,
  SUM(VALUE_AT_RISK_USD) as REVENUE_AT_STAKE,
  AVG(STOCKOUT_RATE_PCT) as AVG_STOCKOUT_RATE,  -- already 0-100 scale
  SUM(UNITS_AT_RISK) as TOTAL_UNITS_AT_RISK
```
